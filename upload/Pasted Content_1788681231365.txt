function hl(src){
  const re = /(\/\/[^\n]*)|('(?:[^'\\\n]|\\.)*'|"(?:[^"\\\n]|\\.)*")|\b(const|let|var|function|return|import|from|export|default|if|else|for|while|new|class|extends|async|await|try|catch|typeof|interface|type|of|in|null|true|false)\b|\b(\d+(?:\.\d+)?)\b/g;
  let out='', last=0, m;
  while((m = re.exec(src))){
    out += esc(src.slice(last, m.index));
    const cls = m[1]?'tok-c':m[2]?'tok-s':m[3]?'tok-k':'tok-n';

    last = m.index + m[0].length;
  }

}

/* ============================================================
   Split-text engine (from the GSAP demo, retuned for this theme)
   data-split="words|letters"  data-dir="up|down|left|right"
   data-shift data-blur data-dur data-stagger
   [data-split-manual] — split, but driven by an external timeline
============================================================ */
let splitTriggers = [];

function splitTargets(el, mode){
  const out = [];
  const walk = (node)=>{
    [...node.childNodes].forEach(ch=>{
      if(ch.nodeType===3){
        const txt = ch.textContent;
        if(!txt) return;
        const frag = document.createDocumentFragment();
        txt.split(/(\s+)/).forEach(tok=>{
          if(!tok) return;
          if(/^\s+$/.test(tok)){ frag.appendChild(document.createTextNode(' ')); return; }
          const w = document.createElement('span'); w.className='st-word';
          if(mode==='letters'){
            for(const c of tok){
              const l=document.createElement('span'); l.className='st-letter'; l.textContent=c;
              w.appendChild(l); out.push(l);
            }
          } else { w.textContent = tok; out.push(w); }
          frag.appendChild(w);
        });
        node.replaceChild(frag, ch);
      } else if(ch.nodeType===1 && ch.tagName!=='BR'){
        walk(ch); /* keep <br> and nested spans (.muted, brackets) intact */
      }
    });
  };
  walk(el);
  return out;
}

function initSplitAuto(scope){
  if(!HAS_GSAP() || RM) return;
  gsap.registerPlugin(ScrollTrigger);
  $$('[data-split]:not([data-split-manual])', scope||app).forEach(el=>{
    const mode   = el.dataset.split || 'words';
    const dir    = el.dataset.dir || 'up';
    const shift  = parseFloat(el.dataset.shift)  || (mode==='letters' ? 12 : 26);
    const blur   = parseFloat(el.dataset.blur)   || (mode==='letters' ? 4  : 9);
    const dur    = parseFloat(el.dataset.dur)    || (mode==='letters' ? .45 : .75);
    const stag   = el.dataset.stagger!==undefined ? parseFloat(el.dataset.stagger) : (mode==='letters' ? .012 : .07);
    const targets = splitTargets(el, mode);
    if(!targets.length) return;
    const axis = (dir==='left'||dir==='right') ? 'x' : 'y';
    const from = (dir==='down'||dir==='left') ? -shift : shift;
    gsap.set(targets, {opacity:0, filter:'blur('+blur+'px)', [axis]:from});
    gsap.to(targets, {
      opacity:1, filter:'blur(0px)', [axis]:0,
      duration:dur, ease:'power2.out', stagger:stag,
      scrollTrigger:{trigger:el, start:'top 86%', once:true}
    });
  });
  /* collect this route's triggers so the next route can kill them */
  splitTriggers = ScrollTrigger.getAll();
}

function killSplits(){
  if(window.ScrollTrigger) ScrollTrigger.getAll().forEach(st=>st.kill());
  splitTriggers = [];
}

      case 'code': return codeBlock(b.lang, b.x);
      case 'quote': return `<blockquote><p>${b.x}</p>${b.by?`<cite>— ${b.by}</cite>`:''}</blockquote>`;
      case 'img': return `<figure class="gsc art-cover" style="margin:26px 0"><img src="${seedImg(b.seed,900,480)}" alt="${b.alt}" loading="lazy"><figcaption>${b.cap||''}</figcaption></figure>`;
      case 'diagram': return diagramSVG(b.nodes, b.cap);
      case 'table': return `<table class="art-table"><thead><tr>${b.head.map(h=>`<th>${h}</th>`).join('')}</tr></thead><tbody>${b.rows.map(r=>`<tr>${r.map(c=>`<td>${c}</td>`).join('')}</tr>`).join('')}</tbody></table>`;
      default: return `<p>${b.x}</p>`;
    }
  }).join('');
}

const revealObs = new IntersectionObserver(es=>{
  es.forEach(e=>{ if(e.isIntersecting){ e.target.classList.add('in'); revealObs.unobserve(e.target); } });
},{threshold:.1, rootMargin:'0px 0px -6% 0px'});

const cntObs = new IntersectionObserver(es=>{
  es.forEach(e=>{ if(e.isIntersecting){ countUp(e.target); cntObs.unobserve(e.target); } });
},{threshold:.4});
function bindCounters(){ $$('[data-count]',app).forEach(el=>cntObs.observe(el)); }
function countUp(el){
  const end=parseFloat(el.dataset.count), suf=el.dataset.suffix||'';
  const t0=performance.now(), dur=900;
  (function f(t){
    const k=Math.min(1,(t-t0)/dur), e=1-Math.pow(1-k,3);
    el.textContent=Math.round(end*e)+suf;
    if(k<1) requestAnimationFrame(f);
  })(t0);
}

/* ============================================================
   Glyph field (playground demo)
============================================================ */
class GlyphField{
  constructor(canvas,o={}){
    this.cv=canvas; this.ctx=canvas.getContext('2d');
    this.o=Object.assign({gap:30,size:13,base:.05,near:.9,radius:170,rgb:null},o);
    this.chars='{}<>/;:=()[]#*&+—%'.split('');
    this.pts=[]; this.mx=-1e4; this.my=-1e4; this.rip=[];
    this.run=true;
    this.build();
    this._resize=()=>this.build();
    addEventListener('resize',this._resize);
    const host=this.cv.parentElement;
    this._move=e=>{ const r=this.cv.getBoundingClientRect(); this.mx=e.clientX-r.left; this.my=e.clientY-r.top; };
    this._leave=()=>{ this.mx=this.my=-1e4; };
    this._down=e=>{ const r=this.cv.getBoundingClientRect(); this.rip.push({x:e.clientX-r.left,y:e.clientY-r.top,t:performance.now()}); };
    host.addEventListener('pointermove',this._move);
    host.addEventListener('pointerleave',this._leave);
    host.addEventListener('pointerdown',this._down);
    this.io=new IntersectionObserver(es=>{ this.vis=es[0].isIntersecting; });
    this.io.observe(this.cv);
    const loop=t=>{ if(!this.run) return; if(this.vis!==false) this.frame(t); this.raf=requestAnimationFrame(loop); };
    if(RM){ this.frame(0); } else { this.raf=requestAnimationFrame(loop); }
    this.cleanup=()=>{
      this.run=false; cancelAnimationFrame(this.raf);
      removeEventListener('resize',this._resize); this.io.disconnect();
      host.removeEventListener('pointermove',this._move);
      host.removeEventListener('pointerleave',this._leave);
      host.removeEventListener('pointerdown',this._down);
    };
  }
  build(){
    const dpr=Math.min(devicePixelRatio||1,2);
    const w=this.cv.clientWidth||this.cv.parentElement.clientWidth;
    const h=this.cv.clientHeight||this.cv.parentElement.clientHeight;
    if(!w||!h) return;
    this.w=w; this.h=h;
    this.cv.width=w*dpr; this.cv.height=h*dpr;
    this.ctx.setTransform(dpr,0,0,dpr,0,0);
    this.pts=[];
    for(let y=this.o.gap/2;y<h;y+=this.o.gap)
      for(let x=this.o.gap/2;x<w;x+=this.o.gap)
        this.pts.push({x,y,c:this.chars[(Math.random()*this.chars.length)|0],p:Math.random()*6.28});
    if(RM) this.frame(0);
  }
  frame(t){
    const {ctx}=this, o=this.o;
    if(!this.w) return;
    ctx.clearRect(0,0,this.w,this.h);
    ctx.font='400 '+o.size+'px "JetBrains Mono", monospace';
    ctx.textAlign='center'; ctx.textBaseline='middle';
    const col=o.rgb||GLYPH_RGB;
    const now=performance.now();
    this.rip=this.rip.filter(r=>now-r.t<900);
    for(const p of this.pts){
      let a=o.base;
      if(!RM) a+=.028*(Math.sin(t*.0009+p.p*3)+1)/2;
      const dx=p.x-this.mx, dy=p.y-this.my;
      const inf=Math.max(0,1-Math.hypot(dx,dy)/o.radius);
      a+=inf*inf*.8;
      if(inf>.5 && Math.random()<.02) p.c=this.chars[(Math.random()*this.chars.length)|0];
      for(const r of this.rip){
        const age=(now-r.t)/900, rr=age*o.radius*1.7, band=50;
        const dv=Math.abs(Math.hypot(p.x-r.x,p.y-r.y)-rr);
        if(dv<band) a+=(1-dv/band)*(1-age)*.6;
      }
      if(a>.02){
        ctx.fillStyle='rgba('+col+','+Math.min(a,o.near).toFixed(3)+')';
        ctx.fillText(p.c,p.x,p.y);
      }
    }
  }
}

/* ============================================================
   Sphere background — the scroll-driven particle globe
============================================================ */
const SphereBG = (function(){
  const canvas = document.getElementById('sphere-canvas');
  const noop = { refresh(){}, setRoute(){}, kick(){}, pause(){}, resume(){} };
  if(!canvas) return noop;
  const ctx = canvas.getContext('2d');
  if(!ctx) return noop;

  const reduced = matchMedia('(prefers-reduced-motion: reduce)').matches;

  const POS_X = [
    { t: 0, v: 0 },
    { t: 0.22, v: 1.4 * 60 },
    { t: 0.45, v: -0.4 * 60 },
    { t: 0.6, v: 1.2 * 60 },
    { t: 0.7, v: -1.7 * 60 },
    { t: 0.85, v: -1.7 * 60 },
    { t: 1, v: 0 }
  ];
  const POS_Y = [
    { t: 0, v: 0 },
    { t: 0.22, v: -0.2 * 40 },
    { t: 0.45, v: 0.1 * 40 },
    { t: 0.6, v: -0.1 * 40 },
    { t: 0.7, v: 0.05 * 40 },
    { t: 0.85, v: 0.05 * 40 },
    { t: 1, v: 0.2 * 40 }
  ];
  const ROT_Y = [
    { t: 0, v: 0 },
    { t: 0.22, v: Math.PI * 0.55 },
    { t: 0.45, v: Math.PI * 1.05 },
    { t: 0.6, v: Math.PI * 1.3 },
    { t: 0.85, v: Math.PI * 1.3 },
    { t: 1, v: Math.PI * 1.6 }
  ];
  const ROT_X = [
    { t: 0, v: 0 },
    { t: 0.22, v: 0.3 },
    { t: 0.45, v: 0.1 },
    { t: 0.6, v: 0 },
    { t: 0.85, v: 0 },
    { t: 1, v: 0.1 }
  ];
  const SCALE = [
    { t: 0, v: 1 },
    { t: 0.22, v: 0.92 },
    { t: 0.45, v: 0.95 },
    { t: 0.6, v: 1.05 },
    { t: 0.85, v: 1.05 },
    { t: 1, v: 0.55 }
  ];

  function trackValue(kfs, p){
    if(p <= kfs[0].t) return kfs[0].v;
    for(let i=1;i<kfs.length;i++){
      if(p <= kfs[i].t){
        const a=kfs[i-1], b=kfs[i];
        const f=(p-a.t)/(b.t-a.t);
        return a.v+(b.v-a.v)*f;
      }
    }
    return kfs[kfs.length-1].v;
  }
  const lerp = (a,b,t)=>a+(b-a)*t;
  function mixC(c1,c2,t,out){
    out[0]=c1[0]+(c2[0]-c1[0])*t;
    out[1]=c1[1]+(c2[1]-c1[1])*t;
    out[2]=c1[2]+(c2[2]-c1[2])*t;
    return out;
  }

  const STOPS_DARK  = [[255,255,255],[167,171,178],[108,112,117],[21,22,26],[10,10,10]];
  const STOPS_LIGHT = [[204,208,214],[167,171,178],[108,112,117],[26,28,34],[12,12,14]];
  function rampAt(st,t,out){
    if(t < .28) return mixC(st[0], st[1], t/.28, out);
    if(t < .55) return mixC(st[1], st[2], (t-.28)/.27, out);
    if(t < .8)  return mixC(st[2], st[3], (t-.55)/.25, out);
    return mixC(st[3], st[4], Math.min(1,(t-.8)/.2), out);
  }

  const BUCKETS = 48, ASTEPS = 16;
  const lut = []; for(let i=0;i<BUCKETS;i++) lut.push([0,0,0]);
  const styleCache = new Map();
  let lutK = -9;
  const tmpD=[0,0,0], tmpL=[0,0,0];
  function rebuildLUT(k){
    for(let i=0;i<BUCKETS;i++){
      const t=i/(BUCKETS-1);
      rampAt(STOPS_DARK, t, tmpD);
      rampAt(STOPS_LIGHT, t, tmpL);
      const c=lut[i];
      c[0]=(tmpD[0]+(tmpL[0]-tmpD[0])*k)|0;
      c[1]=(tmpD[1]+(tmpL[1]-tmpD[1])*k)|0;
      c[2]=(tmpD[2]+(tmpL[2]-tmpD[2])*k)|0;
    }
    styleCache.clear(); lutK = k;
  }

  const PERSPECTIVE = 3.4;
  const LAT_STEPS = 108, LON_STEPS = 136;

  const particles = [];
  for(let i=0;i<LAT_STEPS;i++){
    const theta = -Math.PI/2 + (i/(LAT_STEPS-1))*Math.PI;
    const y0 = -Math.sin(theta);
    const ring = Math.cos(theta);
    for(let j=0;j<LON_STEPS;j++){
      const phi = (j/LON_STEPS)*Math.PI*2 - Math.PI;
      particles.push({ x0: ring*Math.cos(phi), y0, z0: ring*Math.sin(phi) });
    }
  }

  const drawPool = new Array(particles.length);
  for(let i=0;i<drawPool.length;i++) drawPool[i] = { px:0, py:0, z:0, bi:0, ai:0, size:0 };
  const sortPool = [];

  let W=0, H=0;
  const IDLE_SPIN = reduced ? 0 : 0.0016;
  const MAX_FLICK_SPIN = 0.05;
  let rotX = -0.3, rotY = 0.62, velX = 0, velY = IDLE_SPIN;
  let dragging = false, lastPX = 0, lastPY = 0, activePointer = null;
  let paused = false, rafId = 0;
  let targetProgress = 0, smoothProgress = 0, kSmooth = 0, lastTime = 0;
  let isHome = true;

  function onScroll(){
    const max = Math.max(0, document.documentElement.scrollHeight - innerHeight);
    targetProgress = max > 0 ? Math.min(1, Math.max(0, scrollY/max)) : 0;
    scheduleFrame();
  }

  const resize = () => {
    const DPR = Math.min(devicePixelRatio||1, 2);
    W = innerWidth; H = innerHeight;
    canvas.width = Math.max(1, Math.round(W*DPR));
    canvas.height = Math.max(1, Math.round(H*DPR));
    canvas.style.width = W+'px';
    canvas.style.height = H+'px';
    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
    scheduleFrame();
  };

  const EXCLUDE = 'a, button, input, textarea, select, label, summary, .hd, footer, .drawer, .scrim, .ovl, .toasts, .hero-title, .hero-info, .hero-stats, .hero-ctas, .pc, .fp-media, .b-card, .j-card, .t-card, .tst-card, .svc, .proc, .code, .dg-fig, .art-table, .faq-item, .l-rows, .mx-rows, .vt, .tl, .sb-grid, .metrics, .resp, .feat-block, .chs-row, .gallery, .pfeat, .con-info, .c-panel, .sheet, .news, .toc, .author, .takeaway, .cmt, .disc, .term, .pop-rows, .pg-card, .pg-demo, .resume-actions';

  function pointerDown(x,y){
    dragging = true;
    lastPX = x; lastPY = y;
  }
  function pointerMove(x,y){
    if(!dragging) return;
    const dx = x-lastPX, dy = y-lastPY;
    velY = Math.max(-MAX_FLICK_SPIN, Math.min(MAX_FLICK_SPIN, -dx*0.0045));
    velX = Math.max(-MAX_FLICK_SPIN, Math.min(MAX_FLICK_SPIN, -dy*0.0045));
    rotY += velY; rotX += velX;
    rotX = Math.max(-1.3, Math.min(1.3, rotX));
    lastPX = x; lastPY = y;
    scheduleFrame();
  }
  function pointerUp(){ dragging = false; }
  function onPointerDown(e){
    if((e.target) && e.target.closest && e.target.closest(EXCLUDE)) return;
    if(activePointer !== null) return;
    activePointer = e.pointerId;
    pointerDown(e.clientX, e.clientY);
  }
  function onPointerMove(e){
    if(e.pointerId !== activePointer) return;
    pointerMove(e.clientX, e.clientY);
  }
  function releasePointer(e){
    if(e.pointerId !== activePointer) return;
    activePointer = null;
    pointerUp();
  }

  const frame = (time) => {
    rafId = 0;
    if(paused) return;

    const dt = lastTime ? Math.min(0.1, (time-lastTime)/1000) : 0.016;
    lastTime = time;

    if(reduced){ smoothProgress = targetProgress; }
    else { smoothProgress += (targetProgress-smoothProgress)*(1-Math.exp(-dt/0.35)); }

    const p = smoothProgress;
    const posX = trackValue(POS_X, p);
    const posY = trackValue(POS_Y, p);
    const rotCoreY = trackValue(ROT_Y, p);
    const rotCoreX = trackValue(ROT_X, p);
    const scale = trackValue(SCALE, p);

    if(!dragging){
      velY += (IDLE_SPIN - velY)*0.02;
      rotY += velY;
      velX *= 0.94;
      rotX += velX;
      rotX = Math.max(-1.3, Math.min(1.3, rotX));
    }

    const siteDark = document.documentElement.dataset.theme === 'dark';
    const kT = siteDark ? 0
      : (isHome ? Math.min(1, Math.max(0, (p-0.004)/0.11)) : 1);
    if(reduced) kSmooth = kT;
    else kSmooth += (kT - kSmooth)*(1-Math.exp(-dt/0.45));
    if(Math.abs(kSmooth - lutK) > 0.004) rebuildLUT(kSmooth);
    const alphaMul = lerp(1, 0.72, kSmooth);
    const sizeMul = lerp(1, 0.88, kSmooth);

    const finalRotX = rotX + rotCoreX;
    const finalRotY = rotY + rotCoreY;
    const R = Math.min(W,H)*0.335*scale;
    const cosX = Math.cos(finalRotX), sinX = Math.sin(finalRotX);
    const cosY = Math.cos(finalRotY), sinY = Math.sin(finalRotY);
    const cx = W*0.5 + posX;
    const cy = H*0.5 + posY;
    const sizeScale = W/720;

    ctx.clearRect(0,0,W,H);

    let count = 0;
    for(let i=0;i<particles.length;i++){
      const pt = particles[i];
      const x1 = pt.x0*cosY - pt.z0*sinY;
      const z1 = pt.x0*sinY + pt.z0*cosY;
      const y1 = pt.y0*cosX - z1*sinX;
      const z2 = pt.y0*sinX + z1*cosX;
      if(z2 < 0.02) continue;

      const factor = PERSPECTIVE/(PERSPECTIVE - z2);
      const d = drawPool[count++];
      d.px = cx + x1*R*factor;
      d.py = cy + y1*R*factor;
      d.z = z2;
      const latT = Math.max(0, Math.min(1, (1+y1)/2));
      d.bi = Math.min(BUCKETS-1, (latT*BUCKETS)|0);
      const a = (0.35 + 0.25*factor)*alphaMul;
      d.ai = Math.max(0, Math.min(ASTEPS-1, (a*ASTEPS)|0));
      d.size = 1.1*factor*sizeScale*sizeMul;
    }

    sortPool.length = count;
    for(let i=0;i<count;i++) sortPool[i] = drawPool[i];
    sortPool.sort((a,b)=>a.z-b.z);
    for(let i=0;i<count;i++){
      const d = sortPool[i];
      const key = d.bi*ASTEPS + d.ai;
      let s = styleCache.get(key);
      if(!s){
        const c = lut[d.bi];
        s = 'rgba('+c[0]+','+c[1]+','+c[2]+','+(d.ai/ASTEPS).toFixed(3)+')';
        styleCache.set(key, s);
      }
      ctx.fillStyle = s;
      ctx.fillRect(d.px-d.size, d.py-d.size, d.size*2, d.size*2);
    }
    if(!reduced) rafId = requestAnimationFrame(frame);
  };

  const scheduleFrame = () => {
    if(reduced && !rafId && !paused) rafId = requestAnimationFrame(frame);
  };
  const onVisibilityChange = () => {
    if(document.hidden){
      paused = true;
      if(rafId){ cancelAnimationFrame(rafId); rafId = 0; }
    } else {
      paused = false;
      lastTime = 0;
      if(!rafId) rafId = requestAnimationFrame(frame);
    }
  };

  resize();
  onScroll();
  window.addEventListener('resize', resize);
  window.addEventListener('scroll', onScroll, {passive:true});
  document.addEventListener('pointerdown', onPointerDown);
  document.addEventListener('pointermove', onPointerMove);
  document.addEventListener('pointerup', releasePointer);
  document.addEventListener('pointercancel', releasePointer);
  document.addEventListener('visibilitychange', onVisibilityChange);
  if(!rafId) rafId = requestAnimationFrame(frame);
  requestAnimationFrame(()=>canvas.classList.add('on'));

  return {
    setRoute(path){ isHome = path==='/' || path===''; },
    refresh(){ onScroll(); },
    kick(){ scheduleFrame(); },
    pause(){ paused = true; if(rafId){ cancelAnimationFrame(rafId); rafId = 0; } },
    resume(){ if(!paused) return; paused = false; lastTime = 0; if(!rafId) rafId = requestAnimationFrame(frame); }
  };
})();

/* ============================================================
   404 terminal
============================================================ */
function initTerminal(path){
  const body=$('#term-body'); if(!body) return;
  const input=$('#t-in-input'), inRow=$('#t-in');
  const CMD={home:'#/',start:'#/',about:'#/about',projects:'#/projects',experience:'#/experience',
    blog:'#/blog',resume:'#/resume',contact:'#/contact',skills:'#/skills',services:'#/services',
    uses:'#/uses',playground:'#/playground',testimonials:'#/testimonials'};
  const timers=[];
  const addLine=(txt,cls)=>{
    const d=document.createElement('div');
    d.className='t-line'+(cls?' '+cls:''); d.textContent=txt;
    body.appendChild(d); body.scrollTop=body.scrollHeight; return d;
  };
  const queue=[
    {c:'cmd', x:"route --resolve '"+path+"'"},
    {c:'err', x:'404 — component not found'},
    {c:'out', x:"the page you requested isn't part of this build."},
    {c:'out', x:'no cached fallback. no redirect. just this shell.'},
    {c:'out', x:"type 'help' for routes that exist."}
  ];
  let qi=0, ci=0, cur=null;
  function step(){
    if(qi>=queue.length){ inRow.style.display='flex'; if(input) input.focus(); return; }
    const item=queue[qi];
    if(!cur){ cur=addLine(item.c==='cmd'?'$ ':'', item.c); ci=0; }
    if(ci<item.x.length){ cur.textContent+=item.x[ci++]; timers.push(setTimeout(step,13)); body.scrollTop=body.scrollHeight; }
    else { cur=null; qi++; timers.push(setTimeout(step,200)); }
  }
  timers.push(setTimeout(step,350));

  function runCmd(v){
    v=(v||'').trim().toLowerCase();
    if(!v) return;
    addLine('$ '+v,'cmd');
    if(v==='clear'){ body.innerHTML=''; return; }
    if(v==='help'){
      addLine('available commands:','out');
      addLine('  home · projects · blog · about · experience','out');
      addLine('  skills · services · resume · uses · contact','out');
      addLine('  clear — wipe this shell','out');
      return;
    }
    if(CMD[v]){ addLine('→ resolving '+CMD[v]+' …','out'); timers.push(setTimeout(()=>{ location.hash=CMD[v]; },420)); return; }
    addLine("command not found: "+v+" — try 'help'",'err');
  }
  if(input) input.addEventListener('keydown',e=>{
    if(e.key==='Enter'){ runCmd(input.value); input.value=''; }
  });
  document.addEventListener('term-run',e=>{
    if(input){ $('#t-in').style.display='flex'; input.focus();
      input.value=e.detail;
      input.dispatchEvent(new KeyboardEvent('keydown',{key:'Enter',bubbles:true}));
    }
  });
  cleanupFns.push(()=>{ timers.forEach(clearTimeout); });
}

let palOpen=false, palFlat=[], palSel=0;
function buildCmdIndex(){
  const items=[];
  const pages=[['Home','#/'],['About','#/about'],['Projects','#/projects'],['Skills','#/skills'],
    ['Experience','#/experience'],['Services','#/services'],['Résumé','#/resume'],['Blog','#/blog'],
    ['Playground','#/playground'],['Uses','#/uses'],['Testimonials','#/testimonials'],['Contact','#/contact']];
  pages.forEach(p=>items.push({g:'Pages',t:p[0],s:'page',go:p[1]}));
  PROJECTS.forEach(p=>items.push({g:'Projects',t:p.name,s:p.cat.toLowerCase(),go:'#/projects/'+p.slug}));
  POSTS.forEach(p=>items.push({g:'Writing',t:p.title,s:p.cat.toLowerCase(),go:'#/blog/'+p.slug}));
  items.push({g:'Actions',t:'Copy email address',s:'delinb23@gmail.com',act:'copy'});
  items.push({g:'Actions',t:'Download résumé (PDF)',s:'print',go:'#/resume',act:'print'});
  items.push({g:'Actions',t:'Toggle dark mode',s:'theme',act:'theme'});
  return items;
}
function renderCmdList(q){
  q=(q||'').toLowerCase();
  const all=buildCmdIndex();
  const hit=q? all.filter(i=>(i.t+' '+i.s).toLowerCase().includes(q)) : all;
  palFlat=hit; palSel=0;
  const list=$('#cp-list');
  if(!hit.length){ list.innerHTML='<p style="padding:22px;font-size:12px;color:var(--ink-faint)">No matches — try “project”, “blog”, “resume”…</p>'; return; }
  let html='', lastG='';
  hit.forEach((it,i)=>{
    if(it.g!==lastG){ html+='<p class="cp-group">'+it.g+'</p>'; lastG=it.g; }
    html+='<button class="cp-item'+(i===0?' on':'')+'" data-pi="'+i+'"><b>'+it.t+'</b><span>'+it.s+'</span></button>';
  });
  list.innerHTML=html;
}
function moveSel(d){
  if(!palFlat.length) return;
  palSel=(palSel+d+palFlat.length)%palFlat.length;
  $$('.cp-item').forEach(el=>el.classList.toggle('on', +el.dataset.pi===palSel));
  const on=$('.cp-item.on'); if(on) on.scrollIntoView({block:'nearest'});
}
function openPalette(){ palOpen=true; $('#cmdk').classList.add('on'); const q=$('#cp-q'); q.value=''; renderCmdList(''); setTimeout(()=>q.focus(),30); }
function closePalette(){ palOpen=false; $('#cmdk').classList.remove('on'); $('#cp-q').blur(); }
function goCmd(it){
  closePalette();
  if(it.act==='copy'){ copyText(EMAIL,'Email copied — talk soon.'); return; }
  if(it.act==='theme'){ toggleTheme(); return; }
  location.hash=it.go;
  if(it.act==='print') setTimeout(()=>window.print(),450);
}
